```python
class PsychologicalTriggers:
    def __init__(self):
        self.triggers = {
            'scarcity': self.apply_scarcity,
            'social_proof': self.apply_social_proof,
            'authority': self.apply_authority,
            'reciprocity': self.apply_reciprocity,
            'commitment': self.apply_commitment,
            'loss_aversion': self.apply_loss_aversion
        }
        self.ethical_checker = EthicalBoundaryChecker()
    
    def enhance_prompt(self, base_prompt, context):
        # Select relevant triggers
        relevant_triggers = self.select_triggers(context)
        
        # Apply triggers ethically
        enhanced = base_prompt
        for trigger in relevant_triggers:
            if self.ethical_checker.is_ethical(trigger, context):
                enhanced = self.triggers[trigger](enhanced, context)
        
        # Optimize cognitive load
        enhanced = self.optimize_cognitive_load(enhanced)
        
        return enhanced
    
    def apply_scarcity(self, prompt, context):
        if context.get('limited_availability'):
            prompt += "\n\n⏰ Limited time: This opportunity expires soon."
        return prompt
    
    def apply_social_proof(self, prompt, context):
        if context.get('user_count'):
            prompt += f"\n\n👥 Join {context['user_count']} others who have succeeded."
        return prompt
```
